function restaurarLista() { const input = prompt("Pega aquí el texto guardado:"); if (input) { try { // Limpia espacios y saltos de línea extraños const cleanInput = input.trim().replace(/\s+/g, ' '); const list = JSON.parse(cleanInput); window.playlistItems = list; alert(`✅ Restaurados ${list.length} canales.`); location.reload(); } catch(e) { alert("❌ Error: Asegúrate de pegar todo el texto exactamente como lo copiaste."); } } }
